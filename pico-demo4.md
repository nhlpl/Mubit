def rule90_step(state: int, width: int = 32) -> int:
    """
    Applies one step of Rule 90 to a 32-bit integer state.
    Rule 90: new[i] = left[i-1] XOR right[i+1]
    
    In bitwise terms:
        new = (state << 1) XOR (state >> 1)
    
    With boundary conditions (cyclic wrap for true Rule 90 on a ring).
    """
    left_shift = (state << 1) & ((1 << width) - 1)  # Shift left, mask
    right_shift = state >> 1                        # Shift right
    new_state = left_shift ^ right_shift
    return new_state & ((1 << width) - 1)


def rule90_step_cyclic(state: int, width: int = 32) -> int:
    """
    Rule 90 on a cyclic ring (wraparound boundaries).
    This preserves the Sierpinski fractal purity.
    """
    left_shift = ((state << 1) | (state >> (width - 1))) & ((1 << width) - 1)
    right_shift = ((state >> 1) | (state << (width - 1))) & ((1 << width) - 1)
    return left_shift ^ right_shift


def rule90_step_arm_optimized(state: int, width: int = 32) -> int:
    """
    ARM Cortex-M0+ optimized Rule 90:
        EORS R2, R0, R0, LSL #1    (state ^ (state << 1))
        EORS R0, R2, R0, LSR #1    ( ... ^ (state >> 1))
    This is 2 instructions per step (plus masking).
    On the Pico, this executes in ~2 cycles per step.

        This is the Monster's 3-transposition in the collapsed space.

        This is exactly the Rule-90 update on the 16-bit ring.

        self.gate_count += 1
        # Phase flips according to parity of popcount (fermionic sign)
        if self.basis.bit_count() & 1:
            self.phase = -self.phase

    # -------------------------------------------------------------------------

        """Applies a lightweight XOR-shift to the basis index."""

    # -------------------------------------------------------------------------

        """Multiplies the basis by the golden ratio (mod 65536)."""
        self.basis = (self.basis * 0x9E37) & 0xFFFF
